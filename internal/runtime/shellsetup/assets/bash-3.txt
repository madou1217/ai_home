# aih-shell-options-v1: {"completion":true,"typo":true,"wrapper":false,"historyTail":1000}
# aih 终端集成：TAB 补全 + 命令打错时的纠正建议。
# 由 `aih shell apply` 生成，请勿手改；调整用 `aih shell apply <选项>`，移除用 `aih shell remove`。
AIH_SHELL_BIN='__AIH_SHELL_BIN__'
AIH_SHELL_TYPO=1
AIH_SHELL_WRAPPER=0
AIH_SHELL_HISTORY_TAIL=1000

_aih_shell_ready() {
  if [ -n "$AIH_SHELL_BIN" ] && [ -x "$AIH_SHELL_BIN" ]; then return 0; fi
  AIH_SHELL_BIN="$(command -v aih 2>/dev/null)"
  [ -n "$AIH_SHELL_BIN" ]
}

_aih_shell_history() {
  history "$AIH_SHELL_HISTORY_TAIL" 2>/dev/null
}

_aih_shell_reason_label() {
  case "${1:-}" in
    learned) printf '%s' '你确认过' ;;
    case) printf '%s' '大小写' ;;
    transposition) printf '%s' '两个字母写反了' ;;
    typo) printf '%s' '打错字符' ;;
    abbrev) printf '%s' '少写了字母' ;;
    prefix) printf '%s' '前缀匹配' ;;
    history) printf '%s' '历史里用过' ;;
    *) printf '%s' '拼写接近' ;;
  esac
}

# ---- TAB 补全 ----
_aih_complete() {
  _aih_shell_ready || return 1
  local _aih_cur _aih_i _aih_line _aih_out _aih_value _aih_prefix _aih_quote _aih_hint
  _aih_hint=0
  local -a _aih_words
  COMPREPLY=()
  type compopt >/dev/null 2>&1 && compopt +o default
  # Bash 4.4+：保持后端给的优先级顺序（命令在前、历史在后）；3.2 没有 compopt，只能按字典序显示。
  type compopt >/dev/null 2>&1 && compopt -o nosort 2>/dev/null
  for (( _aih_i=1; _aih_i<=COMP_CWORD; _aih_i++ )); do
    _aih_words+=("${COMP_WORDS[_aih_i]}")
  done
  # Bash splits --flag=value around =; keep the protocol words canonical.
  if (( ${#_aih_words[@]} >= 2 )) && [[ "${_aih_words[${#_aih_words[@]}-1]}" == "=" ]]; then
    _aih_words[${#_aih_words[@]}-1]=""
  elif (( ${#_aih_words[@]} >= 3 )) && [[ "${_aih_words[${#_aih_words[@]}-2]}" == "=" ]]; then
    _aih_value="${_aih_words[${#_aih_words[@]}-1]}"
    unset "_aih_words[${#_aih_words[@]}-1]"
    _aih_words[${#_aih_words[@]}-1]="$_aih_value"
  fi
  [ "${#_aih_words[@]}" -eq 0 ] && _aih_words=("")
  _aih_out="$(_aih_shell_history | "$AIH_SHELL_BIN" __complete --history-stdin --hints -- "${_aih_words[@]}" 2>/dev/null)"
  while IFS= read -r _aih_line; do
    [ -z "$_aih_line" ] && continue
    _aih_value="${_aih_line%%$'\x1f'*}"
    _aih_group="${_aih_line##*$'\x1f'}"
    if [ "$_aih_group" = "提示" ]; then
      _aih_hint=1
      continue
    fi
    # Bash 3.2 keeps --flag=value in COMP_WORDS but readline replaces only
    # the value after its = word break. Do not insert the flag twice.
    if [[ "$COMP_WORDBREAKS" == *"="* && "${COMP_WORDS[COMP_CWORD]}" == --*=* ]]; then
      _aih_prefix="${COMP_WORDS[COMP_CWORD]%%=*}="
      [[ "$_aih_value" == "$_aih_prefix"* ]] && _aih_value="${_aih_value#"$_aih_prefix"}"
    fi
    COMPREPLY+=("$_aih_value")
  done <<< "$_aih_out"
  # 无 AIH 候选时交回 readline；命令位仍不混入文件名。
  if (( ${#COMPREPLY[@]} == 0 && COMP_CWORD > 1 )) && [ "$_aih_hint" -eq 0 ]; then
    if type compopt >/dev/null 2>&1; then
      compopt -o default
    else
      # macOS 自带 Bash 3.2 没有 compopt，保留路径及空格转义。
      while IFS= read -r _aih_line; do
        [ -z "$_aih_line" ] && continue
        [[ -d "$_aih_line" ]] && _aih_line="${_aih_line%/}/"
        case "${COMP_WORDS[COMP_CWORD]}" in
          \"*)
            _aih_cur="${_aih_line//\\/\\\\}"
            _aih_cur="${_aih_cur//\$/\\\$}"
            _aih_cur="${_aih_cur//\`/\\\`}"
            _aih_cur="${_aih_cur//\"/\\\"}" ;;
          \'*)
            _aih_quote="'\\''"
            _aih_cur="${_aih_line//\'/${_aih_quote}}" ;;
          *) printf -v _aih_cur "%q" "$_aih_line" ;;
        esac
        COMPREPLY+=("$_aih_cur")
      done <<< "$(compgen -f -- "${COMP_WORDS[COMP_CWORD]}")"
    fi
  fi
  return 0
}
complete -F _aih_complete aih

# ---- 命令打错时的纠正建议 ----
if [ "${_AIH_BASH_HOOK_INSTALLED:-0}" != 1 ] && declare -F command_not_found_handle >/dev/null 2>&1 && ! declare -F _aih_prev_command_not_found_handle >/dev/null 2>&1; then
  if [[ "$(declare -f command_not_found_handle)" != *"_aih_shell_fallback_not_found"* ]]; then
    eval "$(declare -f command_not_found_handle | sed '1s/command_not_found_handle/_aih_prev_command_not_found_handle/')"
  fi
fi

_aih_shell_fallback_not_found() {
  if declare -F _aih_prev_command_not_found_handle >/dev/null 2>&1; then
    _aih_prev_command_not_found_handle "$@"
    return $?
  fi
  printf "%s\n" "bash: $1: command not found" >&2
  return 127
}

# 两个入口共用同一套「列候选 → 确认 → 运行/放进历史」流程：
#   _aih_shell_typo_offer        命令本身不存在（ahi、gti）
#   _aih_shell_typo_offer_inside aih 存在但紧随其后的词写错（aih codx）
# 参数：提示语 / 显示用的命令行 / 其余原样交给 __suggest。返回 0 = 已处理。
_aih_shell_typo_offer_impl() {
  local _aih_msg="$1" _aih_head="$2"
  shift 2
  _aih_shell_ready || return 1
  local _aih_out _aih_kind_line _aih_typo_key _aih_line _aih_display _aih_reason _aih_argv_text _aih_answer
  local -a _aih_displays _aih_argvs _aih_reasons _aih_argv
  _aih_out="$(_aih_shell_history | "$AIH_SHELL_BIN" __suggest --history-stdin -- "$@" 2>/dev/null)"
  [ -z "$_aih_out" ] && return 1
  _aih_kind_line="${_aih_out%%$'\n'*}"
  _aih_typo_key="${_aih_kind_line##*$'\x1f'}"
  case "${_aih_kind_line#*$'\x1f'}" in none*) return 1 ;; esac
  [ -z "$_aih_typo_key" ] && _aih_typo_key="$_aih_head"
  while IFS= read -r _aih_line; do
    case "$_aih_line" in aih:cand$'\x1f'*) ;; *) continue ;; esac
    _aih_line="${_aih_line#aih:cand$'\x1f'}"
    _aih_display="${_aih_line%%$'\x1f'*}"
    _aih_line="${_aih_line#*$'\x1f'}"
    _aih_reason="${_aih_line%%$'\x1f'*}"
    _aih_argv_text="${_aih_line#*$'\x1f'}"
    _aih_displays+=("$_aih_display")
    _aih_argvs+=("$_aih_argv_text")
    _aih_reasons+=("$_aih_reason")
  done <<< "$_aih_out"
  [ "${#_aih_displays[@]}" -eq 0 ] && return 1
  printf "\n" >&2
  printf "[aih] %s，你是不是想运行：\n" "$_aih_msg" >&2
  local _aih_i
  for (( _aih_i=0; _aih_i<${#_aih_displays[@]}; _aih_i++ )); do
    printf "  %s) %s  (%s)\n" "$(( _aih_i + 1 ))" "${_aih_displays[_aih_i]}" "$(_aih_shell_reason_label "${_aih_reasons[_aih_i]}")" >&2
  done
  printf "选择序号运行 [1-%s]（回车=取消，e=放进历史自己编辑）：" "${#_aih_displays[@]}" >&2
  read -r _aih_answer || return 1
  if [ "$_aih_answer" = "e" ] || [ "$_aih_answer" = "E" ]; then
    history -s -- "${_aih_displays[0]}"
    printf "已放入历史：%s（按 ↑ 编辑后回车运行）\n" "${_aih_displays[0]}" >&2
    return 0
  fi
  case "$_aih_answer" in ""|*[!0-9]*) return 0 ;; esac
  [ "$_aih_answer" -ge 1 ] && [ "$_aih_answer" -le "${#_aih_displays[@]}" ] || return 0
  IFS=$'\x1f' read -r -a _aih_argv <<< "${_aih_argvs[$(( _aih_answer - 1 ))]}"
  [ "${#_aih_argv[@]}" -eq 0 ] && return 0
  "$AIH_SHELL_BIN" __learn "$_aih_typo_key" -- "${_aih_argv[@]}" >/dev/null 2>&1
  printf "[aih] 运行：%s\n" "${_aih_displays[$(( _aih_answer - 1 ))]}" >&2
  "${_aih_argv[@]}"
  return 0
}

_aih_shell_typo_offer() {
  _aih_shell_typo_offer_impl "没有 $1 这个命令" "$1" "$@"
}

_aih_shell_typo_offer_inside() {
  # 错的是紧随 `aih` 之后的那个词：把完整命令行交给 __suggest --inside。
  _aih_shell_typo_offer_impl "aih 不认识 ${1:-}" "aih ${1:-}" --inside aih "$@"
}

_AIH_BASH_HOOK_INSTALLED=1
command_not_found_handle() {
  if [ -t 0 ] && [ -t 1 ] && [ "${AIH_SHELL_TYPO:-0}" = "1" ]; then
    _aih_shell_typo_offer "$@" && return 0
  fi
  _aih_shell_fallback_not_found "$@"
}

# aih-shell-options-v1: {"completion":true,"typo":true,"wrapper":false,"historyTail":1000}
# aih 终端集成：TAB 补全 + 命令打错时的纠正建议。
# 由 `aih shell apply` 生成，请勿手改；调整用 `aih shell apply <选项>`，移除用 `aih shell remove`。
$global:AIH_SHELL_BIN = '__AIH_SHELL_BIN__'
$global:AIH_SHELL_TYPO = $true
$global:AIH_SHELL_WRAPPER = $false
$global:AIH_SHELL_HISTORY_TAIL = 1000
$global:_AIH_SHELL_US = [char]0x1f

function global:Get-AihShellReady {
    if ($global:AIH_SHELL_BIN -and (Test-Path -LiteralPath $global:AIH_SHELL_BIN)) { return $true }
    $found = Get-Command aih -ErrorAction SilentlyContinue
    if (-not $found) { return $false }
    $global:AIH_SHELL_BIN = $found.Source
    return [bool]$global:AIH_SHELL_BIN
}

function global:Get-AihShellHistory {
    $items = @(Get-History | Select-Object -Last $global:AIH_SHELL_HISTORY_TAIL | ForEach-Object { $_.CommandLine })
    return $items
}

function global:Get-AihShellReasonLabel {
    param([string]$Reason)
    switch ($Reason) {
        'learned' { '你确认过' }
        'case' { '大小写' }
        'transposition' { '两个字母写反了' }
        'typo' { '打错字符' }
        'abbrev' { '少写了字母' }
        'prefix' { '前缀匹配' }
        'history' { '历史里用过' }
        default { '拼写接近' }
    }
}

# 调一次 aih 的轻量入口；返回原始输出行（不抛出）。
function global:Invoke-AihShellTool {
    param([string[]]$ToolArgs, [string[]]$HistoryLines)
    try {
        if (-not (Get-AihShellReady)) { return @() }
        $historyText = ($HistoryLines -join "`n")
        # Windows PowerShell 5.1 按 OEM 代码页解码原生命令输出：UTF-8 的中文说明会把
        # 换行字节一起吞掉，候选只剩一两条。调用期间临时切到 UTF-8。
        $prevOut = [Console]::OutputEncoding
        try {
            [Console]::OutputEncoding = [System.Text.Encoding]::UTF8
            $out = $historyText | & $global:AIH_SHELL_BIN @ToolArgs 2>$null
        } finally {
            [Console]::OutputEncoding = $prevOut
        }
        return @($out)
    } catch { return @() }
}

# ---- TAB 补全 ----
Register-ArgumentCompleter -Native -CommandName aih -ScriptBlock {
    param($wordToComplete, $commandAst, $cursorPosition)
    if (-not (Get-AihShellReady)) { return }
    try {
        $elements = @($commandAst.CommandElements | ForEach-Object { $_.Extent.Text })
        $all = @()
        if ($elements.Count -gt 1) { $all = @($elements[1..($elements.Count - 1)]) }
        $prior = @()
        # PS 5.1 对 native 命令在“空格后”仍把上一个词当作 $wordToComplete，
        # 直接用会把 `aih server ` 补成顶层前缀。按光标前一个字符判断是否在词边界。
        $boundary = $false
        $text = [string]$commandAst.Extent.Text
        if ($cursorPosition -gt $text.Length) {
            $boundary = $true
        } elseif ($cursorPosition -gt 0 -and [char]::IsWhiteSpace($text[$cursorPosition - 1])) {
            $boundary = $true
        }
        if ($boundary) {
            $prior = $all
            $wordToComplete = ""
        } elseif ($all.Count -ge 1 -and $all[-1] -eq $wordToComplete) {
            if ($all.Count -ge 2) { $prior = @($all[0..($all.Count - 2)]) }
        } else {
            $prior = $all
        }
        $toolArgs = @("__complete", "--shell", "powershell", "--history-stdin", "--current=$wordToComplete", "--") + $prior
        $lines = Invoke-AihShellTool -ToolArgs $toolArgs -HistoryLines (Get-AihShellHistory)
        foreach ($line in $lines) {
            if (-not $line) { continue }
            $parts = $line -split [string]$global:_AIH_SHELL_US
            $value = $parts[0]
            $desc = ""
            if ($parts.Count -ge 2 -and $null -ne $parts[1]) { $desc = [string]$parts[1] }
            if ([string]::IsNullOrEmpty($desc)) { $desc = ' ' }
            $label = $value
            if ($desc -ne ' ') { $label = $value + '  ' + $desc }
            [System.Management.Automation.CompletionResult]::new($value, $label, "ParameterValue", $desc)
        }
    } catch { }
}

function global:New-AihSwitchResult {
    param([string]$Value, [string]$Description)
    $label = $Value
    if (-not [string]::IsNullOrEmpty($Description)) { $label = $Value + '  ' + $Description }
    if ([string]::IsNullOrEmpty($Description)) { $Description = ' ' }
    return [System.Management.Automation.CompletionResult]::new($Value, $label, 'ParameterName', $Description)
}
Register-ArgumentCompleter -Native -CommandName powershell,powershell.exe -ScriptBlock {
    param($wordToComplete, $commandAst, $cursorPosition)
    $prev = $null
    if ($commandAst.CommandElements.Count -ge 2) { $prev = $commandAst.CommandElements[-2].Extent.Text }
    if ($prev -eq '-ExecutionPolicy') {
        foreach ($v in @('Restricted','AllSigned','RemoteSigned','Unrestricted','Bypass','Undefined')) {
            if ($v -like "$wordToComplete*") { New-AihSwitchResult $v '执行策略' }
        }
        return
    }
    if ($prev -eq '-InputFormat' -or $prev -eq '-OutputFormat') {
        foreach ($v in @('Text','XML')) { if ($v -like "$wordToComplete*") { New-AihSwitchResult $v '格式' } }
        return
    }
    if ($prev -eq '-WindowStyle') {
        foreach ($v in @('Normal','Hidden','Minimized','Maximized')) { if ($v -like "$wordToComplete*") { New-AihSwitchResult $v '窗口样式' } }
        return
    }
    if (-not $wordToComplete.StartsWith('-')) { return }
    foreach ($item in $global:_AIH_POWERSHELL_SWITCHES) {
        if ($item[0] -like "$wordToComplete*") { New-AihSwitchResult $item[0] $item[1] }
    }
}

Register-ArgumentCompleter -Native -CommandName cmd,cmd.exe -ScriptBlock {
    param($wordToComplete, $commandAst, $cursorPosition)
    if ($wordToComplete -match '^/(E|F|V):') {
        $prefix = $Matches[0]
        foreach ($v in @('ON','OFF')) {
            $candidate = $prefix + $v
            if ($candidate -like "$wordToComplete*") { New-AihSwitchResult $candidate '开关取值' }
        }
        return
    }
    if ($wordToComplete -match '^/T:') { return }
    if (-not $wordToComplete.StartsWith('/')) { return }
    foreach ($item in $global:_AIH_CMD_SWITCHES) {
        if ($item[0] -like "$wordToComplete*") { New-AihSwitchResult $item[0] $item[1] }
    }
}

# ---- aih 的 Tab 用菜单展示候选（仅当 Tab 仍是默认的循环补全）----
# PowerShell 默认 TabCompleteNext 只在命令行里静默轮换；让 aih 用户看到的「候选效果」
# 更接近 zsh 的菜单。这里只接管 aih 行，其它命令保持原行为；用户自定义过 Tab 则不动。
if (Get-Command Set-PSReadLineKeyHandler -ErrorAction SilentlyContinue) {
    try {
        $tabHandler = @(Get-PSReadLineKeyHandler -Bound | Where-Object { $_.Key -contains 'Tab' }) | Select-Object -First 1
        $tabFunction = ''
        if ($tabHandler) { $tabFunction = [string]$tabHandler.Function }
        if ($tabFunction -eq 'TabCompleteNext' -or $tabFunction -eq 'Complete') {
            $global:_AIH_TAB_FUNCTION = $tabFunction
            Set-PSReadLineKeyHandler -Chord Tab -ScriptBlock {
                param($key, $arg)
                $line = $null
                $cursor = $null
                [Microsoft.PowerShell.PSConsoleReadLine]::GetBufferState([ref]$line, [ref]$cursor)
                if ($line -match '^\s*(&\s*)?\S*aih(\.exe)?(\s|$)') {
                    [Microsoft.PowerShell.PSConsoleReadLine]::MenuComplete($key, $arg)
                } elseif ($global:_AIH_TAB_FUNCTION -eq 'Complete') {
                    [Microsoft.PowerShell.PSConsoleReadLine]::Complete($key, $arg)
                } else {
                    [Microsoft.PowerShell.PSConsoleReadLine]::TabCompleteNext($key, $arg)
                }
            }
        }
    } catch { }
}

# ---- 裸 - / -- 的兜底 ----
# PS 5.1 的 TabExpansion2 对「当前词正好是 - 或 --」直接短路，不会调用 native
# completer；这里包一层，只在这种形态下自己查 aih，其余原样交给系统实现。
if (-not (Get-Command _aih_original_tabexpansion2 -ErrorAction SilentlyContinue)) {
    $aihOriginalTabExpansion = Get-Command TabExpansion2 -CommandType Function -ErrorAction SilentlyContinue
    if ($aihOriginalTabExpansion) {
        Set-Item -Path function:global:_aih_original_tabexpansion2 -Value $aihOriginalTabExpansion.ScriptBlock
    }
}
if (Get-Command _aih_original_tabexpansion2 -ErrorAction SilentlyContinue) {
    function global:TabExpansion2 {
        param($inputScript, $cursorColumn = $inputScript.Length)
        $line = $inputScript.Substring(0, [Math]::Min($cursorColumn, $inputScript.Length))
        if ($line -match '^(?<pre>.*)\s(?<token>-{1,2})$') {
            $token = $Matches['token']
            $words = @($Matches['pre'].TrimEnd() -split '\s+')
            $leaf = Split-Path -Leaf $words[0]
            if ($leaf -eq 'aih' -or $leaf -eq 'aih.exe') {
                $prior = @()
                if ($words.Count -gt 1) { $prior = $words[1..($words.Count - 1)] }
                $toolArgs = @('__complete', '--shell', 'powershell', '--history-stdin', "--current=$token", '--') + $prior
                $lines = Invoke-AihShellTool -ToolArgs $toolArgs -HistoryLines (Get-AihShellHistory)
                $results = New-Object 'System.Collections.ObjectModel.Collection[System.Management.Automation.CompletionResult]'
                foreach ($item in $lines) {
                    if (-not $item) { continue }
                    $parts = $item -split [string]$global:_AIH_SHELL_US
                    $desc = ''
                    if ($parts.Count -ge 2 -and $null -ne $parts[1]) { $desc = [string]$parts[1] }
                    if ([string]::IsNullOrEmpty($desc)) { $desc = ' ' }
                    $label = $parts[0]
                    if ($desc -ne ' ') { $label = $parts[0] + '  ' + $desc }
                    $results.Add([System.Management.Automation.CompletionResult]::new($parts[0], $label, 'ParameterValue', $desc))
                }
                $start = $line.Length - $token.Length
                return [System.Management.Automation.CommandCompletion]::new($results, 0, $start, $token.Length)
            } elseif ($leaf -in @('powershell','powershell.exe','pwsh','pwsh.exe')) {
                $switches = $global:_AIH_POWERSHELL_SWITCHES
                if ($leaf -like 'pwsh*') { $switches = $global:_AIH_PWSH_SWITCHES }
                $results = New-Object 'System.Collections.ObjectModel.Collection[System.Management.Automation.CompletionResult]'
                foreach ($item in $switches) {
                    if ($item[0] -like "$token*") { $results.Add((New-AihSwitchResult $item[0] $item[1])) }
                }
                $start = $line.Length - $token.Length
                return [System.Management.Automation.CommandCompletion]::new($results, 0, $start, $token.Length)
            }
        }
        $current = ''
        $prefixLine = $line.TrimEnd()
        if ($line -eq $prefixLine) {
            $parts = @($line -split '\s+')
            if ($parts.Count -gt 0) {
                $current = $parts[-1]
                $prefixLine = $line.Substring(0, $line.Length - $current.Length).TrimEnd()
            }
        }
        $parts = @($prefixLine -split '\s+' | Where-Object { $_ -ne '' })
        if ($parts.Count -gt 0) {
            $shellLeaf = Split-Path -Leaf $parts[0]
            if ($shellLeaf -in @('powershell','powershell.exe','pwsh','pwsh.exe')) {
                $prev = ''
                if ($parts.Count -ge 2) { $prev = $parts[-1] }
                $values = $null
                $valueDesc = ''
                switch ($prev) {
                    '-ExecutionPolicy' { $values = @('Restricted','AllSigned','RemoteSigned','Unrestricted','Bypass','Undefined'); $valueDesc = '执行策略' }
                    '-InputFormat' { $values = @('Text','XML'); $valueDesc = '格式' }
                    '-OutputFormat' { $values = @('Text','XML'); $valueDesc = '格式' }
                    '-WindowStyle' { $values = @('Normal','Hidden','Minimized','Maximized'); $valueDesc = '窗口样式' }
                }
                $results = New-Object 'System.Collections.ObjectModel.Collection[System.Management.Automation.CompletionResult]'
                if ($null -ne $values) {
                    foreach ($v in $values) { if ($v -like "$current*") { $results.Add((New-AihSwitchResult $v $valueDesc)) } }
                } elseif ($current.StartsWith('-')) {
                    $switches = $global:_AIH_POWERSHELL_SWITCHES
                    if ($shellLeaf -like 'pwsh*') { $switches = $global:_AIH_PWSH_SWITCHES }
                    foreach ($item in $switches) {
                        if ($item[0] -like "$current*") { $results.Add((New-AihSwitchResult $item[0] $item[1])) }
                    }
                } else {
                    return _aih_original_tabexpansion2 $inputScript $cursorColumn
                }
                $start = $line.Length - $current.Length
                return [System.Management.Automation.CommandCompletion]::new($results, 0, $start, $current.Length)
            }
        }
        return _aih_original_tabexpansion2 $inputScript $cursorColumn
    }
}

# ---- 命令打错时的纠正建议（PSReadLine 回车钩子）----
# 取证结论：PSReadLine 的 -CommandValidationHandler 对「命令不存在」根本不会被调用
# （BasicEditing.cs 里的内建 CommandNotFound 检查先短路返回），所以只能接管 Enter 键：
# 先判一次「这个命令存在吗」，不存在才列候选；其余情况原样调用 AcceptLine()。
function global:Show-AihShellTypoSuggestions {
    param([string]$Message, [string]$Head, [string[]]$ToolArgs)
    $lines = Invoke-AihShellTool -ToolArgs $ToolArgs -HistoryLines (Get-AihShellHistory)
    if ($lines.Count -eq 0) { return $null }
    $kind = ($lines[0] -split [string]$global:_AIH_SHELL_US)
    if ($kind.Count -lt 2 -or $kind[1] -eq "none") { return $null }
    $typoKey = $Head
    if ($kind.Count -ge 3 -and $kind[2]) { $typoKey = $kind[2] }
    $candidates = @()
    foreach ($line in $lines[1..($lines.Count - 1)]) {
        $parts = $line -split [string]$global:_AIH_SHELL_US
        if ($parts.Count -lt 4 -or $parts[0] -ne "aih:cand") { continue }
        $candidates += [pscustomobject]@{
            Display = $parts[1]
            Reason = (Get-AihShellReasonLabel $parts[2])
            Argv = @($parts[3] -split [string]$global:_AIH_SHELL_US)
        }
    }
    if ($candidates.Count -eq 0) { return $null }
    Write-Host ""
    Write-Host "[aih] $Message，你是不是想运行：" -ForegroundColor Yellow
    for ($i = 0; $i -lt $candidates.Count; $i++) {
        Write-Host ("  {0}) {1}  ({2})" -f ($i + 1), $candidates[$i].Display, $candidates[$i].Reason) -ForegroundColor DarkGray
    }
    $answer = Read-Host ("选择序号运行 [1-{0}]（回车=取消，e=放到命令行自己编辑）" -f $candidates.Count)
    if (-not $answer) { return $null }
    if ($answer -eq "e" -or $answer -eq "E") { return $candidates[0].Display }
    if ($answer -notmatch "^\d+$") { return $null }
    $index = [int]$answer
    if ($index -lt 1 -or $index -gt $candidates.Count) { return $null }
    $chosen = $candidates[$index - 1]
    & $global:AIH_SHELL_BIN __learn $typoKey -- @($chosen.Argv) 2>$null | Out-Null
    Write-Host ("[aih] 运行：{0}" -f $chosen.Display) -ForegroundColor DarkGray
    if ($chosen.Argv.Count -gt 1) { & $chosen.Argv[0] @($chosen.Argv[1..($chosen.Argv.Count - 1)]) } else { & $chosen.Argv[0] }
    return ""
}

function global:Set-AihShellEnterHandler {
    if (-not (Get-Command Set-PSReadLineKeyHandler -ErrorAction SilentlyContinue)) { return }
    if (-not $global:AIH_SHELL_TYPO) { return }
    try {
        Set-PSReadLineKeyHandler -Key Enter -ScriptBlock {
            param($key, $arg)
            try {
                $line = ""
                $cursor = 0
                [Microsoft.PowerShell.PSConsoleReadLine]::GetBufferState([ref]$line, [ref]$cursor)
                if ($global:AIH_SHELL_TYPO -and $line -and $line.Trim() -and $line.Length -lt 500 -and (Get-AihShellReady)) {
                    $trimmed = $line.Trim()
                    $name = ($trimmed -split "\s+")[0]
                    if ($name -match "^[A-Za-z0-9_\-]+$" -and -not (Get-Command $name -ErrorAction SilentlyContinue)) {
                        $tokens = $trimmed -split "\s+"
                        $rest = @()
                        if ($tokens.Count -gt 1) { $rest = @($tokens[1..($tokens.Count - 1)]) }
                        $toolArgs = @("__suggest", "--history-stdin", "--") + @($name) + $rest
                        $corrected = Show-AihShellTypoSuggestions -Message "没有 $name 这个命令" -Head $name -ToolArgs $toolArgs
                        if ($null -ne $corrected) {
                            if ($corrected) {
                                [Microsoft.PowerShell.PSConsoleReadLine]::Replace(0, $line.Length, $corrected)
                            } else {
                                [Microsoft.PowerShell.PSConsoleReadLine]::Replace(0, $line.Length, "")
                            }
                            return
                        }
                    }
                }
            } catch { }
            [Microsoft.PowerShell.PSConsoleReadLine]::AcceptLine()
        }
    } catch { }
}
